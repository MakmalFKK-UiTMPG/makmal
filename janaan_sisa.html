// ✉️ FUNGSI UNTUK MENGHANTAR EMEL AUTOMATIK VIA EMAILJS async function hantarNotifikasiEmel(dataSisa) { let teksKomposisi = dataSisa.komposisi_sisa.map(k => `${k.nama}: ${k.peratus}`).join(', '); const templateParams = { // 🎯 E-mel penerima yang telah ditetapkan to_email: 'nurha1559@uitm.edu.my', nama_pemohon: dataSisa.nama_pemohon, no_id: dataSisa.no_id, no_tel: dataSisa.no_tel, nama_makmal: dataSisa.nama_makmal, nama_penyelia: dataSisa.nama_penyelia, kategori_pengguna: dataSisa.kategori_pengguna, fizikal_sisa: dataSisa.fizikal_sisa, tarikh_dijana: dataSisa.tarikh_penstoran, berat_kasar: dataSisa.berat_kasar, komposisi_sisa: teksKomposisi }; try { // ⚠️ Pastikan SERVICE_ID dan TEMPLATE_ID dimasukkan mengikut akaun EmailJS anda await emailjs.send('YOUR_SERVICE_ID', 'YOUR_TEMPLATE_ID', templateParams); console.log("✅ E-mel notifikasi berjaya dihantar ke nurha1559@uitm.edu.my!"); } catch (err) { console.error("❌ Gagal hantar e-mel notifikasi:", err); } }
